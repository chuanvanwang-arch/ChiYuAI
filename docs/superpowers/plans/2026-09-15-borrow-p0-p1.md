// src/config/signalFreshness.js
// P0-1（设计 §3.1）：信号时间衰减档（配置驱动；出厂默认即为「无衰减=向后兼容」下限）
export const DEFAULT_AGE_TIERS = Object.freeze([
  { max_days: 7, multiplier: 1.0 },
  { max_days: 30, multiplier: 0.6 },
  { max_days: 90, multiplier: 0.3 },
  { max_days: null, multiplier: 0.1 },
]);

// 纯函数：给定信号年龄（天）→ 衰减系数（[0,1]）。
//   tsMs 为空（信号无时间戳）→ 1.0（向后兼容，不衰减）
//   tiers 为空 → 1.0
export function freshnessMultiplier(ageDays, tiers = DEFAULT_AGE_TIERS) {
  if (ageDays === null || ageDays === undefined) return 1.0;
  if (!Array.isArray(tiers) || tiers.length === 0) return 1.0;
  for (const t of tiers) {
    if (t.max_days === null || ageDays <= t.max_days) return t.multiplier ?? 1.0;
  }
  return 1.0;
}

// 纯函数：ts 时间戳 → 天数（负数兜底 0）
export function ageDaysOf(ts, now = Date.now()) {
  if (!ts) return null;
  const t = new Date(ts).getTime();
  if (Number.isNaN(t)) return null;
  return Math.max(0, Math.floor((now - t) / 86400000));
}
```

// src/config/prospectingRules.js（改动 computeFitScore；mergeProspectingRules 增 signal_age_tiers 合并）
import { freshnessMultiplier, ageDaysOf } from './signalFreshness.js';

// mergeProspectingRules 内（在 candidate_limit 之前加）：
if (Array.isArray(tenantCfg.signal_age_tiers)) out.signal_age_tiers = structuredClone(tenantCfg.signal_age_tiers);

// computeFitScore 改（对每个命中信号按新鲜度加权）：
export function computeFitScore(candidate, rules) {
  const signals = candidate.signals || {};
  const w = rules.signals || {};
  const tiers = rules.signal_age_tiers || null;
  const sigTs = {
    funding: (candidate.signals ?? {})?.funding_ts,
    hiring: (candidate.signals ?? {})?.hiring_ts,
    tender: (candidate.signals ?? {})?.tender_ts,
    social: (candidate.signals ?? {})?.social_ts,
  };
  let num = 0, den = 0;
  for (const [k, weight] of Object.entries(w)) {
    if (typeof weight !== 'number') continue;
    den += weight;
    if (signals[k]) {
      const mult = tiers ? freshnessMultiplier(ageDaysOf(sigTs[k]), tiers) : 1.0;
      num += weight * mult;
    }
  }
  return den > 0 ? num / den : 0;
}
```

// src/sales/decisionChainValidator.js
// P0-2（设计 §3.2）：决策链角色漂移校验器。
// 纯函数、零 DB、零粒子改动；映射词表可配置（fail-open：未知角色/无模板 → 不误报）。
export const DEFAULT_ROLE_TEMPLATE = Object.freeze({
  required: ['veto', 'budget', 'approve'],
  champion: { required: false },
});

export const DEFAULT_ROLE_MAP = Object.freeze([
  { role: 'budget', keywords: ['总经理', '总裁', '副总', '财务', 'CFO', 'CEO'] },
  { role: 'approve', keywords: ['总经理', '总裁', '副总', 'CXO', 'CIO'] },
  { role: 'veto', keywords: ['采购', '法务', '合规', '生产', 'quality', '审计'] },
  { role: 'recommend', keywords: ['研发', '技术', 'IT', '工程师', '产品'] },
  { role: 'champion', keywords: ['研发', '产品', '技术', '市场'] },
]);

export function validateDecisionChain(contacts, ctx = {}, opts = {}) {
  const template = opts.template || DEFAULT_ROLE_TEMPLATE;
  const roleMap = opts.roleMap || DEFAULT_ROLE_MAP;
  const required = Array.isArray(template.required) ? template.required : [];
  const list = Array.isArray(contacts) ? contacts : [];

  const matchRole = (c) => {
    const hay = `${c.title || ''} ${c.department || ''} ${c.job_title || ''}`.toLowerCase();
    for (const m of roleMap) {
      if (m.role && m.keywords.some((k) => hay.includes(k.toLowerCase()))) return m.role;
    }
    return null;
  };
  const assigned = list.map((c) => ({ contact: c, role: c.decision_power || matchRole(c) }));
  const covered = new Set(assigned.map((a) => a.role).filter(Boolean));
  const missingRoles = required.filter((r) => !covered.has(r));
  // 漂移：挂了 decision_power 但 title/department 不匹配该角色的 map
  const driftContacts = assigned
    .filter((a) => a.contact.decision_power && matchRole(a.contact) !== a.contact.decision_power)
    .map((a) => ({ title: a.contact.title, department: a.contact.department, declared: a.contact.decision_power, inferred: matchRole(a.contact) }));
  const completeness = required.length ? covered.size / required.length : 1.0;
  return { completeness, missingRoles, driftContacts, assigned };
}
```

// P0-3（设计 §3.3）：富集成本/命中率聚合（纯函数，抗假绿：分母恒真实）。
export function computeEnrichmentRate({ enriched = 0, withContacts = 0 } = {}) {
  const denominator = enriched > 0 ? enriched : 0;
  const rate = denominator > 0 ? withContacts / denominator : 0;
  return { rate, hits: withContacts, denominator };
}

export function aggregateEnrichment(records = []) {
  const totalCost = records.reduce((s, r) => s + (r.cost || 0), 0);
  const calls = records.reduce((s, r) => s + (r.calls || 0), 0);
  const hits = records.reduce((s, r) => s + (r.hits || 0), 0);
  const { rate } = computeEnrichmentRate({ enriched: calls, withContacts: hits });
  return { totalCost, calls, hits, rate };
}

// src/skills/methodOutreachHook.js
// P1-1b（设计 §4.1）：钩子生成 + 可验证性校验（对齐 CitationGuard 纪律）。
// 只读（不写）；DEAL payload.outreach_hooks[] 由调用方（Action/前端）经第 0 闸写入。
import { freshnessMultiplier, ageDaysOf } from '../config/signalFreshness.js';
import { computeFitScore } from '../config/prospectingRules.js';

const MAX_WORDS = 30;

export function pickAnchor(signal, deal) {
  if (signal?.url) return { type: 'signal', url: signal.url, label: `${signal.type} 信号` };
  if (deal?.homepage_url) return { type: 'homepage', url: deal.homepage_url, label: '官网' };
  return null; // fail-closed：无锚点不生成
}

export async function generateHook({ deal = {}, signal = null, contact = {}, rules = null, llm = null }) {
  const anchor = pickAnchor(signal, deal);
  if (!anchor) throw new Error('outreach-hook: 无锚点可引用（fail-closed），补充 signal.url 或 deal.homepage_url');
  // 事实校验：钩子内每个事实必须来自 deal/signal/contact（可溯源）
  const facts = [
    { fact: deal.name, source: anchor.label || 'deal', ok: !!deal.name },
    { fact: contact.title, source: 'contact', ok: !!contact.title },
    { fact: signal.type, source: 'signal', ok: !!signal?.type },
  ];
  const verifiable = facts.map((f) => ({ ...f, ok: f.ok ? true : false }));
  const hook = llm
    ? await llm.genHook({ deal, signal, contact, anchor })
    : buildFallbackHook(deal, signal, contact, anchor);
  if (hook.split(' ').length > MAX_WORDS) throw new Error('outreach-hook: 钩子超 30 词');
  return {
    hook, anchor, verifiable,
    verifiable_all: verifiable.every((v) => v.ok),
    confidence: verifiable.every((v) => v.ok) ? 0.9 : 0.4,
    freshness: freshnessMultiplier(ageDaysOf(signal?.ts), rules?.signal_age_tiers),
  };
}

export function buildFallbackHook(deal, signal, contact, anchor) {
  const who = contact.title ? `作为${contact.title}` : '';
  const what = signal?.type ? `注意到贵司${signal.type}动态` : `关注到贵司${deal.name}`;
  return `您好${who}，${what}（${anchor.label}）。我们专注${deal.industry || 'B2B 数字化'}，想介绍相关方案。`.slice(0, MAX_WORDS * 2);
}
```

// P1-3（设计 §4.3）：预热子状态机（DEAL payload.preheat 内；不改 stage 状态机）。
// 铁律：HITL——无人工确认不产生任何对外动作；AI 仅排期与提醒。
export const PREHEAT_STATES = Object.freeze(['waiting', 'scheduled', 'engaged', 'ready']);

const EVENTS = Object.freeze({
  waiting:   { schedule: 'scheduled' },
  scheduled: { engage: 'engaged' },   // engage 需 hitl_confirm:true 参数
  engaged:   { ready: 'ready' },
});

export function preheatTransition(state, event, payload = {}) {
  if (event === 'engage' && !payload.hitl_confirm) throw new Error('preheat: engage 需 HITL 人工确认');
  const next = EVENTS[state]?.[event];
  if (!next) throw new Error(`preheat: 非法转移 ${state} --${event}--> ~`);
  return next;
}

export function buildPreheatPlan(candidates, { signal_fresh_map = null } = {}) {
  // 纯函数：按信号新鲜度排序互动序列（联动 P0-1）
  return candidates
    .map((c, i) => ({ ...c, step: i + 1, due: c.signal_ts ? 'asap' : 'next_cycle' }))
    .sort((a, b) => (a.due === 'asap' ? -1 : 1));
}
