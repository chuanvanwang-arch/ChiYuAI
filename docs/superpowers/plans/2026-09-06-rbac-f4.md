```js
// test/context/scope.sysadmin.test.js — F4 sysadmin write_scope 收敛（方案 C）
import { describe, it, expect, beforeAll, afterAll } from 'vitest';
import { enforceScope, BUSINESS_PARTICLE_TYPES, scopePredicate } from '../../src/context/scope.js';
import { seedProfiles, loadProfile } from '../../src/context/roleProfiles.js';
import { queryWrite } from '../../src/db.js';

// 构造 sysadmin profile（与 roleProfiles.js:26 种子同构）
const sysadminProfile = () => ({
  role_tag: 'sysadmin',
  data_scope: { model: 'all', write_scope: { model: 'governance', exclude_types: BUSINESS_PARTICLE_TYPES } },
});

describe('F4 sysadmin write_scope 收敛（方案 C）', () => {
  it('BUSINESS_PARTICLE_TYPES 含 9 业务粒子且不含 CRM_PERSON', () => {
    for (const t of ['CRM_DEAL','CRM_ACCOUNT','CRM_CONTACT','CRM_TECHNICAL_PROPOSAL',
      'CRM_INVOICE','CRM_PAYMENT_RECORD','CRM_CONTRACT','CRM_QUOTATION','CRM_ORDER']) {
      expect(BUSINESS_PARTICLE_TYPES).toContain(t);
    }
    expect(BUSINESS_PARTICLE_TYPES).not.toContain('CRM_PERSON'); // 治理类须放行，不可误杀
  });

  it('C1: sysadmin data_scope 含 write_scope.governance（读仍 all）', async () => {
    await seedProfiles();
    const p = await loadProfile('sysadmin');
    expect(p).not.toBeNull();
    expect(p.data_scope.model).toBe('all');
    expect(p.data_scope.write_scope.model).toBe('governance');
    expect(p.data_scope.write_scope.exclude_types).toEqual(BUSINESS_PARTICLE_TYPES);
  });

  it('C2: sysadmin 写 CRM_DEAL（业务粒子）跨租户 → scope_violation', async () => {
    const ins = await queryWrite(
      `INSERT INTO crm.particles (id,type,tenant_id,payload,meta) VALUES ($1,$2,$3,$4,$5) ON CONFLICT (id) DO NOTHING`,
      ['f4-deal-x','CRM_DEAL','tenant-b',JSON.stringify({ owner_id: 'someone' }),'{}']);
    const r = await enforceScope({ name: 'data-particle-update' },
      { actor: 'sysadmin', tenantId: 'tenant-a' }, { id: 'f4-deal-x' }, sysadminProfile());
    expect(r.ok).toBe(false);
    expect(r.gate).toBe('scope_violation');
    await queryWrite(`DELETE FROM crm.particles WHERE id=$1`, ['f4-deal-x']);
  });

  it('C2: sysadmin 写 CRM_PERSON（治理类）跨租户 → ok（不误杀用户管理）', async () => {
    await queryWrite(
      `INSERT INTO crm.particles (id,type,tenant_id,payload,meta) VALUES ($1,$2,$3,$4,$5) ON CONFLICT (id) DO NOTHING`,
      ['f4-person-x','CRM_PERSON','tenant-b',JSON.stringify({ owner_id: 'sysadmin' }),'{}']);
    const r = await enforceScope({ name: 'data-particle-update' },
      { actor: 'sysadmin', tenantId: 'tenant-a' }, { id: 'f4-person-x' }, sysadminProfile());
    expect(r.ok).toBe(true);
    await queryWrite(`DELETE FROM crm.particles WHERE id=$1`, ['f4-person-x']);
  });

  it('C2: sysadmin 无 id（列表类写）→ ok（交 scopePredicate）', async () => {
    const r = await enforceScope({ name: 'data-particle-read' },
      { actor: 'sysadmin', tenantId: 'tenant-a' }, {}, sysadminProfile());
    expect(r.ok).toBe(true);
  });

  it('C: sysadmin scopePredicate 读无 clause（跨租户读全量保留）', () => {
    const pred = scopePredicate(sysadminProfile(), 'sysadmin', []);
    expect(pred.clause).toBe('');
  });
});
```

```js
// F4-3：sysadmin 写 token 标记治理写范围；业务写由 executor 第1闸 enforceScope 双闸拒（HTTP+MCP 同源）
```
